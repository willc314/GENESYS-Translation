%  SVG HANDLING -- one switch, read this before you build
%
%  FALSE (default): the converted PDFs in images/ are used. Fast, needs no
%                   flags, and is what Overleaf's free plan can finish inside
%                   its compile limit. Re-run tools/svg2pdf.sh after editing
%                   any diagram so the PDF matches the SVG.
%
%  TRUE:            SVGs in images/ are compiled directly by Inkscape. Needs
%                   -shell-escape, and converting seven large diagrams will
%                   usually exceed the free plan's compile timeout. Use it
%                   only on a paid plan or locally.

%-----------------------------------------------------------------------------
% \umlfigure{basename}{caption}{label}
%
% Pass the file name WITHOUT an extension. Resolution order:
%   images/<name>.pdf   preferred: fast, no shell-escape needed
%   images/<name>.png
%   images/<name>.svg   via the svg package (needs Inkscape + -shell-escape)
%   placeholder         so the document always compiles
%-----------------------------------------------------------------------------
\newcommand{\umlplaceholder}[1]{%
	\fbox{\parbox[c][0.30\textheight][c]{0.92\textwidth}{\centering
			\textcolor{gray}{\large Diagram placeholder}\\[0.6em]
			\textcolor{gray}{\texttt{images/#1.svg}}\\[0.6em]
			\textcolor{gray}{\small Export the diagram to \texttt{images/} to replace this.}}}%
}

\newcommand{\umlfigure}[3]{%
	\begin{figure}[htbp]
		\centering
		\IfFileExists{images/#1.pdf}{%
			\includegraphics[width=\textwidth,height=0.80\textheight,keepaspectratio]{#1.pdf}%
		}{\IfFileExists{images/#1.png}{%
				\includegraphics[width=\textwidth,height=0.80\textheight,keepaspectratio]{#1.png}%
			}{\IfFileExists{images/#1.svg}{%
					\ifsvgready\includesvg[width=\textwidth,height=0.80\textheight,keepaspectratio]{#1}\else\umlplaceholder{#1}\fi
				}{%
					\umlplaceholder{#1}%
				}}}
		\caption{#2}
		\label{#3}
	\end{figure}%
}
